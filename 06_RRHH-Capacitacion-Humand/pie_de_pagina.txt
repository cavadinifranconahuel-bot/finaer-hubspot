<script>
const ch = new Choices('#re', {
  removeItemButton: true,
  searchEnabled: true,
  searchPlaceholderValue: 'Escribí un nombre para buscar...',
  placeholder: true,
  placeholderValue: 'Seleccioná personas...',
  noResultsText: 'Sin resultados',
  noChoicesText: 'No hay más opciones',
  itemSelectText: '',
  shouldSort: false,
  searchFields: ['label']
});

document.getElementById('f').addEventListener('submit', async function(e) {
  e.preventDefault();
  const bt = document.getElementById('bt');
  const erDiv = document.getElementById('er');
  erDiv.style.display = 'none';
  bt.disabled = true;
  bt.textContent = 'Enviando...';

  const em  = document.getElementById('em').value.trim();
  const tu  = document.getElementById('tu').value;
  const sel = ch.getValue(true);

  if (!em || !tu || !sel.length) {
    alert('Completá todos los campos antes de enviar.');
    bt.disabled = false; bt.textContent = 'Inscribir equipo'; return;
  }

  const payload = {
    fields: [
      { name: 'email', value: em },
      { name: 'turno_capacitacion_humand', value: tu },
      ...sel.map(v => ({ name: 'recursos_inscribir_humand', value: v }))
    ],
    context: { pageUri: location.href }
  };

  console.log('Enviando payload:', JSON.stringify(payload));
  try {
    const r = await fetch(
      'https://api.hsforms.com/submissions/v3/integration/submit/145725856/afcadf5d-3a40-4844-b3a4-4df144c9bebb',
      { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(payload) }
    );
    const respText = await r.text();
    console.log('Respuesta HubSpot:', r.status, respText);
    if (r.ok) {
      document.getElementById('f').style.display = 'none';
      document.getElementById('ok').style.display = 'block';
    } else {
      erDiv.textContent = '❌ Error al enviar (status ' + r.status + '): ' + respText.slice(0, 200);
      erDiv.style.display = 'block';
      bt.disabled = false;
      bt.textContent = 'Inscribir equipo';
    }
  } catch(err) {
    console.error('Error fetch:', err);
    erDiv.textContent = '❌ Error de red: ' + err.message;
    erDiv.style.display = 'block';
    bt.disabled = false;
    bt.textContent = 'Inscribir equipo';
  }
});
</script>
